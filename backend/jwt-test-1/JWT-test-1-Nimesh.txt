import java.util.*;
public class OUSL_test_2_Nimesh {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        while (true) { // loop ekk vidiyta auto start venna

            System.out.print("Enter Your Name: ");
            String name = input.nextLine();
            String token = Base64.getEncoder().encodeToString((name + ":secret").getBytes());
            System.out.println("Token: " + token);
            System.out.print("Enter your Token: ");
            String decoded = new String(Base64.getDecoder().decode(input.nextLine()));
            System.out.println(decoded.endsWith(":secret") ? "Hi Welcome, "
                    + decoded.split(":")[0] + "!" : " Invalid Token, Please try again ! ");
            System.out.println("===========================");  // restart patan gnna thana


        }
    }
}
